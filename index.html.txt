    <script>
        function switchTab(dayId) {
            // Hide all tab contents
            const contents = document.querySelectorAll('.tab-content');
            contents.forEach(content => content.classList.add('hidden'));

            // Show selected tab content
            const activeContent = document.getElementById(dayId);
            if(activeContent) {
                activeContent.classList.remove('hidden');
            }

            // Reset tab styles
            const tabs = document.querySelectorAll('.tab-btn');
            tabs.forEach(tab => {
                tab.classList.remove('border-jp-sakura', 'text-jp-sakura', 'font-bold');
                tab.classList.add('border-transparent', 'text-jp-textMuted', 'font-medium');
            });

            // Active current tab
            const currentTab = document.getElementById('tab-' + dayId);
            if(currentTab) {
                currentTab.classList.remove('border-transparent', 'text-jp-textMuted', 'font-medium');
                currentTab.classList.add('border-jp-sakura', 'text-jp-sakura', 'font-bold');
            }
        }

        function updateChecklist() {
            const checkboxes = document.querySelectorAll('input[type="checkbox"]');
            let checkedCount = 0;
            checkboxes.forEach(cb => {
                if(cb.checked) checkedCount++;
            });
            document.getElementById('checklistProgress').innerText = `${checkedCount} / ${checkboxes.length} 完成`;
        }
    </script>
